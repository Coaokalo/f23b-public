cmake_minimum_required(VERSION 3.25)
project(F23B_Weapons LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The weapon connection requires Windows x64")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../config/releases/independent-weapons.json" config)
string(JSON count LENGTH "${config}" binaries)
math(EXPR last "${count} - 1")
set(header "struct Supported { const wchar_t* name; const char* sha; };\nstatic constexpr Supported supported[] = {\n")
foreach(index RANGE ${last})
    string(JSON path MEMBER "${config}" binaries ${index})
    string(JSON hash GET "${config}" binaries "${path}")
    get_filename_component(name "${path}" NAME)
    string(APPEND header "    {L\"${name}\", \"${hash}\"},\n")
endforeach()
string(APPEND header "};\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/supported.h" "${header}")
add_library(F23B_Weapons SHARED Weapons.cpp)
target_include_directories(F23B_Weapons PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(F23B_Weapons PRIVATE cxx_std_17)
target_link_libraries(F23B_Weapons PRIVATE bcrypt)
set_property(TARGET F23B_Weapons PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
